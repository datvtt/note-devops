kubectl create namespace argocd


helm repo add argo https://argoproj.github.io/argo-helm
helm repo update


nano argocd-values.yaml
####################################################
global:
  domain: argocd.showtech.cloud  # sửa lại domain của bạn

server:
  # Service để nội bộ Cluster dùng
  service:
    type: ClusterIP
    port: 80

  # Bật Ingress
  ingress:
    enabled: true
    ingressClassName: nginx   # trùng với ingress-nginx class của bạn

    annotations:
      kubernetes.io/ingress.class: "nginx"
      cert-manager.io/cluster-issuer: "letsencrypt-http"  # tên ClusterIssuer của bạn
      acme.cert-manager.io/http01-edit-in-place: "true"

    hosts:
      - argocd.showtech.cloud  # domain của bạn

    tls:
      - secretName: argocd-tls
        hosts:
          - argocd.showtech.cloud

configs:
  params:
    server.insecure: "true"  # để ingress handle TLS, ArgoCD chạy HTTP nội bộ
##################################################################

helm install argocd argo/argo-cd \
  -n argocd \
  -f argocd-values.yaml
